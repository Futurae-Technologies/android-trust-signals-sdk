object TrustSignalsSDK

Public entry point for the Trust Signals SDK.

Usage:

TrustSignalsSDK.initialize(context, TSConfiguration(appId = "…", sdkId = "…"))

// Coroutine
val collection = TrustSignalsSDK.collect()
val uploaded  = TrustSignalsSDK.collectAndUpload(TSCollectionRequest(accountId = "…", accessToken = "…"))

// Callback
TrustSignalsSDK.collect(onSuccess = { … }, onError = { … })
TrustSignalsSDK.collectAndUpload(TSCollectionRequest(accountId = "…", accessToken = "…"), onSuccess = { … }, onError = { … })

// Scheduled (WorkManager — minimum interval enforced by the OS is 15 minutes)
TrustSignalsSDK.scheduleCollections(
    interval = 15.minutes,
    request  = TSCollectionRequest(accountId = "…", accessToken = "…"),
)
TrustSignalsSDK.stopScheduledCollections()
